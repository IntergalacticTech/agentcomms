Code that speaks for itself

A complete workflow: create inbox, send email, wait for reply, extract OTP.

from freemail import FreeMail

client = FreeMail(api_key="am_live_your_key")

# Create an inbox on any platform domain
inbox = client.inboxes.create(
    display_name="Signup Bot",
    domain="karmascale.net",
)
print(inbox["email"])  # "zk8m2q0rwv3b@karmascale.net"

# Send an email
client.messages.send(
    inbox_id=inbox["id"],
    to=[{"address": "user@example.com"}],
    subject="Hello from my agent",
    body_text="This email was sent by an AI agent.",
)

# Wait for a reply (long-poll, up to 25s)
reply = client.inboxes.wait_for_message(
    inbox_id=inbox["id"],
    sender="user@example.com",
    timeout=25,
)

# Extract an OTP
otp = client.inboxes.extract_otp(
    inbox_id=inbox["id"],
    sender="noreply@service.com",
    timeout=25,
)
print(f"Verification code: {otp['code']}")
import { FreeMail } from 'freemail';

const client = new FreeMail({ apiKey: 'am_live_your_key' });

// Create an inbox on any platform domain
const inbox = await client.inboxes.create({
  display_name: 'Signup Bot',
  domain: 'karmascale.net',
});
console.log(inbox.email);

// Send an email
await client.messages.send(inbox.id, {
  to: [{ address: 'user@example.com' }],
  subject: 'Hello from my agent',
  body_text: 'This email was sent by an AI agent.',
});

// Wait for a reply (long-poll, up to 25s)
const reply = await client.inboxes.waitForMessage(inbox.id, {
  sender: 'user@example.com',
  timeout: 25,
});

// Extract an OTP
const otp = await client.inboxes.extractOtp(inbox.id, {
  sender: 'noreply@service.com',
  timeout: 25,
});
console.log(`Code: ${otp.code}`);

  -H "Content-Type: application/json" \
  -d '{"display_name": "Signup Bot", "domain": "karmascale.net"}'

# Send an email (from an existing inbox)
curl https://api.victorymail.dev/v1/inboxes/INBOX_ID/messages \
  -H "x-api-key: am_live_your_key" \
  -H "Content-Type: application/json" \
  -d '{
    "to": [{"address": "user@example.com"}],
    "subject": "Hello from my agent",
    "body_text": "This email was sent by an AI agent."
  }'

# Wait for a matching reply (long-poll, max 25s)
curl "https://api.victorymail.dev/v1/inboxes/INBOX_ID/wait?sender=user@example.com&timeout=25" \
  -H "x-api-key: am_live_your_key"

# Extract an OTP from an incoming verification email
curl https://api.victorymail.dev/v1/inboxes/INBOX_ID/extract-otp \
  -H "x-api-key: am_live_your_key" \
  -H "Content-Type: application/json" \
  -d '{"sender": "noreply@service.com", "timeout": 25}'
